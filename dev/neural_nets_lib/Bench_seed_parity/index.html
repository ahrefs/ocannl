type verdict =
  1. | Match of {
    1. err : Base.float;
    2. node : Base.string;
    }
    (*

    Within tolerance; err is the worst normalized difference, at node.

    *)
  2. | Mismatch of {
    1. err : Base.float;
    2. node : Base.string;
    }
  3. | Declined of Base.string
    (*

    The candidate compile was refused, typed — a decline the search would have counted.

    *)
  4. | Not_applied
    (*

    The seed never reached its segment: the replay did not find it.

    *)
  5. | Not_perturbed
    (*

    Negative control only: the seeded segment had no statement perturb_code perturbs.

    *)
  6. | Rerendered of Base.string
    (*

    Negative control only (gh-ocannl-1677): the perturbed seed's tensor-core statements did not render as the unperturbed seed's (its mma census differs, or one of the two did not compile), so a mismatch would be caught on a path the seed does not take. A failure.

    *)
  7. | Unjudged of Base.string
    (*

    gh-ocannl-1602: the seed computes on a tf32 arm, but which reference accumulations its tensor-core statements replaced could not be established, so no reference with its rounding exists. A failure, never a silent pass.

    *)
  8. | Failed of Base.string
    (*

    An untyped compile failure, or the run raised.

    *)
type seed_result = {
  1. label : Base.string;
    (*

    The candidate's label as the search prints it (F_sketch[...]).

    *)
  2. conv : Base.bool;
    (*

    The seed is a conv family's (conv_seed).

    *)
  3. verdict : verdict;
  4. tensorized : Base.bool;
    (*

    The routine's mma census says it rendered tensorized.

    *)
  5. mma : Ir.C_syntax.mma_summary;
    (*

    The compiled candidate's mma census (Context.routine.mma); empty when it did not compile.

    *)
  6. compared : Base.int;
    (*

    Nodes compared.

    *)
  7. rounded : Base.string Base.list;
    (*

    gh-ocannl-1602: the nodes whose accumulated products the seed's reference rounds to tf32 (the seed computes them on a tf32 arm); empty when the reference is the plain default pipeline.

    *)
  8. rounded_stmts : Base.int;
    (*

    The reference's accumulation statements so rounded.

    *)
  9. flipped : Base.int;
    (*

    gh-ocannl-1602: the decision cells the seed decided otherwise than its reference; counted only for a seed that mismatched.

    *)
  10. replayed : Base.bool;
    (*

    The verdict is the replayed reference's (flipped cells replayed).

    *)
}
type reference = {
  1. ref_compared : Base.int;
    (*

    The nodes the reference writes, each a comparison candidate.

    *)
  2. ref_restored : Base.int;
    (*

    Read-and-written nodes restored before every run.

    *)
  3. ref_poisoned : Base.int;
    (*

    Written nodes filled with NaN before every run.

    *)
  4. ref_poison_unsafe : Base.string Base.list;
    (*

    Written nodes the poisoned reference left non-finite: never poisoned.

    *)
  5. ref_rounded_compiles : Base.int;
    (*

    gh-ocannl-1602: tf32-rounded references compiled, one per distinct rounded node set in a row.

    *)
  6. ref_replay_compiles : Base.int;
    (*

    References compiled to replay a flipped seed's decisions.

    *)
  7. ref_decisions : Base.string Base.list;
    (*

    The decision nodes (decision_nodes) a replay replays.

    *)
  8. ref_decision_cells : Base.int;
    (*

    Their cells: the replay bound's base.

    *)
  9. ref_unreplayable : Base.string Base.list;
    (*

    Decisions the replay cannot reach.

    *)
}
