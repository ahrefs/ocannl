type verdict =
  1. | Match of {
    1. err : Base.float;
    2. node : Base.string;
    }
    (*

    Within tolerance; err is the worst normalized difference, at node.

    *)
  2. | Mismatch of {
    1. err : Base.float;
    2. node : Base.string;
    }
  3. | Declined of Base.string
    (*

    The candidate compile was refused, typed — a decline the search would have counted.

    *)
  4. | Not_applied
    (*

    The seed never reached its segment: the replay did not find it.

    *)
  5. | Not_perturbed
    (*

    Negative control only: the seeded segment had no Set to perturb.

    *)
  6. | Failed of Base.string
    (*

    An untyped compile failure, or the run raised.

    *)
type seed_result = {
  1. label : Base.string;
    (*

    The candidate's label as the search prints it (F_sketch[...]).

    *)
  2. verdict : verdict;
  3. tensorized : Base.bool;
    (*

    The routine's mma census says it rendered tensorized.

    *)
  4. compared : Base.int;
    (*

    Nodes compared.

    *)
}
type reference = {
  1. ref_compared : Base.int;
    (*

    The nodes the reference writes, each a comparison candidate.

    *)
  2. ref_restored : Base.int;
    (*

    Read-and-written nodes restored before every run.

    *)
  3. ref_poisoned : Base.int;
    (*

    Written nodes filled with NaN before every run.

    *)
  4. ref_poison_unsafe : Base.string Base.list;
    (*

    Written nodes the poisoned reference left non-finite: never poisoned.

    *)
}
